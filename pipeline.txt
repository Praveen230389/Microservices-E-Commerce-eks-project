pipeline {
    agent { label 'jenkins-eks-slave' } 

    environment {
        AWS_ACCOUNT_ID = '457518777735'
        AWS_REGION     = 'us-east-1'
        REGISTRY_URL   = "${AWS_ACCOUNT_ID}.dkr.ecr.${AWS_REGION}.amazonaws.com"
        APP_NAME       = 'cartservice'
        YAML_FILE      = 'cartservice.yaml'
        IMAGE_TAG      = "${BUILD_NUMBER}"
    }

    stages {
        stage('Workspace Cleanup & Checkout') {
            steps {
                cleanWs()
                git branch: 'master', url: "https://github.com"
            }
        }

        // 🛡️ STEP 1: Code Security Audit before compilation
        stage('SonarQube Security Scan') {
            steps {
                script {
                    // Standard enterprise wrapper command to inject tokens securely
                    withSonarQubeEnv('SonarQube-Server') {
                        // Scan only the specific microservice source folder to save time
                        sh "sonar-scanner -Dsonar.projectKey=${APP_NAME} -Dsonar.sources=src/${APP_NAME}/src"
                    }
                }
            }
        }

        // 🛡️ STEP 2: Manifest configuration security check
        stage('IaC Compliance Check (Checkov)') {
            steps {
                script {
                    // Soft-fail ensures pipeline doesn't randomly block on minor warning alerts
                    sh "checkov -f kubernetes-files/${YAML_FILE} --framework kubernetes --soft-fail"
                }
            }
        }

        stage("Build Docker Image") {
            steps {
                script {
                    dir("src/${APP_NAME}/src") {
                        sh "docker build --no-cache -t ${APP_NAME}:latest ."
                    }
                }
            }
        }

        // 🛡️ STEP 3: Binary package patch mapping
        stage("Vulnerability Scan (Trivy)") {
            steps {
                script {
                    // Production rule: If image contains HIGH or CRITICAL issues, stop the pipeline (exit code 1)
                    sh "trivy image --severity HIGH,CRITICAL --exit-code 1 ${APP_NAME}:latest"
                }
            }
        }

        stage("Push Image to AWS ECR") {
            steps {
                script {
                    sh """
                        aws ecr get-login-password --region ${AWS_REGION} | docker login --username AWS --password-stdin ${REGISTRY_URL}
                        docker tag ${APP_NAME}:latest ${REGISTRY_URL}/${APP_NAME}:${IMAGE_TAG}
                        docker push ${REGISTRY_URL}/${APP_NAME}:${IMAGE_TAG}
                    """
                }
            }
        }

        stage('Update Manifest & Git Push') {
            steps {
                dir('kubernetes-files') {
                    withCredentials([string(credentialsId: 'my-git-pattoken', variable: 'git_token')]) {
                        script {
                            sh """
                                git config user.email "Praveenkumar230389@gmail.com"
                                git config user.name "Praveen230389"
                                sed -i "s#image:.*#image: ${REGISTRY_URL}/${APP_NAME}:${IMAGE_TAG}#g" ${YAML_FILE}
                                git add ${YAML_FILE}
                                git commit -m "Automated Production Release: ${APP_NAME}:${IMAGE_TAG} [skip ci]"
                                git push https://${git_token}@://github.com HEAD:master
                            """
                        }
                    }
                }
            }
        }
    }

    // 🛡️ STEP 4: Automated Notification Alerting
    post {
        success {
            slackSend channel: '#alerts-prod-deploy', color: 'good', message: "SUCCESS: ${APP_NAME} build #${IMAGE_TAG} successfully deployed to ECR and Manifest repo updated."
        }
        failure {
            slackSend channel: '#alerts-prod-deploy', color: 'danger', message: "CRITICAL: DevSecOps Gate failed for ${APP_NAME} at build #${IMAGE_TAG}. Review logs immediately."
        }
        always {
            cleanWs()
        }
    }
}
